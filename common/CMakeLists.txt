set(LIB_TARG aroc)

file(GLOB_RECURSE LIB_TARG_SOURCES FOLLOW_SYMLINKS 
    ${CMAKE_CURRENT_SOURCE_DIR}/src/*.c
    ${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/log.c/src/*.c
)

add_library(${LIB_TARG} STATIC ${LIB_TARG_SOURCES})
add_library(${CMAKE_PROJECT_NAME}::${LIB_TARG} ALIAS ${LIB_TARG})

# Required so this .a can be linked into a .so
set_target_properties(${LIB_TARG} PROPERTIES POSITION_INDEPENDENT_CODE ON)

target_include_directories(${LIB_TARG} PUBLIC 
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/log.c/src>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/${LIB_TARG}>
)

target_include_directories(${LIB_TARG} PRIVATE 
    # ${CMAKE_CURRENT_SOURCE_DIR}/mdns
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/mdns>
)

target_compile_options(${LIB_TARG} PRIVATE ${GENERIC_COMPILE_OPTIONS})

target_link_libraries(${LIB_TARG} PRIVATE 
    ${COMMON_LIBRARIES}
    ${CMAKE_PROJECT_NAME}::ucx
)

install(TARGETS ${LIB_TARG}
    EXPORT ${LIB_TARG}Targets
    ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
    INCLUDES DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}
)

# Installation
install(
    DIRECTORY include/ 
    DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/${LIB_TARG}
)
install(
    DIRECTORY log.c/src/
    DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/${LIB_TARG}
    FILES_MATCHING PATTERN "*.h"
)

install(EXPORT ${LIB_TARG}Targets
    FILE ${LIB_TARG}Targets.cmake
    NAMESPACE ${CMAKE_PROJECT_NAME}::
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/${CMAKE_PROJECT_NAME}
)
